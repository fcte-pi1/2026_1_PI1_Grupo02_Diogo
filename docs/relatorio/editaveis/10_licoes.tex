\section{Análise SWOT}















\begin{table}[ht]
\centering
\caption{Análise SWOT do produto.}
\label{tab:swot}

\begin{tabular}{|c|c|c|}
\hline
 & \textbf{Pontos fortes} & \textbf{Pontos fracos} \\ \hline

\parbox{0.1\textwidth}{\centering\textbf{Fatores \\ internos}} &
\parbox{0.4\textwidth}{
\textbf{Forças:}
\begin{itemize}
    \item Projeto compacto, facilitando curvas em labirintos.
    \item Baixo consumo de energia.
    \item Projeto fácil de entender e montar.
\end{itemize}
}
&
\parbox{0.4\textwidth}{
\textbf{Fraquezas:}
\begin{itemize}
    \item Depende da precisão dos sensores para funcionar de maneira eficaz.
    \item Por utilizar solda, o projeto não permite alterações rápidas nos componentes.
    \item A substituição de sensores queimados aumentou um pouco o orçamento previsto.
\end{itemize}
}
\\ \hline

\parbox{0.1\textwidth}{\centering\textbf{Fatores \\ externos}} &
\parbox{0.4\textwidth}{
\textbf{Oportunidades:}
\begin{itemize}
    \item Obter um bom desempenho na competição.
    \item Adicionar futuramente sensores mais precisos para melhorar a navegação.
    \item Reaproveitar o projeto em disciplinas e pesquisas futuras.
    \item Utilizar o conhecimento adquirido em projetos mais complexos.
\end{itemize}
}
&
\parbox{0.4\textwidth}{
\textbf{Ameaças:}
\begin{itemize}
    \item O protótipo não concluir o percurso durante a competição.
    \item Interferências na leitura dos sensores causadas pela iluminação ou pelo ambiente.
    \item Componentes eletrônicos podem sofrer desgaste ou danos.
    \item Falhas de comunicação entre hardware e software podem comprometer o desempenho do protótipo.
\end{itemize}
}
\\ \hline

\end{tabular}
\end{table}
